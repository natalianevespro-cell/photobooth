function generatePhotoStrip() { const ctx = stripCanvas.getContext('2d'); const stripWidth = 560; const stripHeight = 1500; stripCanvas.width = stripWidth; stripCanvas.height = stripHeight; ctx.fillStyle = '#0a0a0a'; ctx.fillRect(0, 0, stripWidth, stripHeight); const borderSize = 10; const availableWidth = stripWidth - (borderSize * 2); const availableHeight = stripHeight - (borderSize * 2); const gap = borderSize; const totalGaps = gap * 3; const imgHeight = (availableHeight - totalGaps) / 4; const imgWidth = availableWidth; let imagesLoaded = 0; capturedDataUrls.forEach((dataUrl, index) => { const img = new Image(); img.onload = () => { const x = borderSize; const y = borderSize + (index * (imgHeight + gap)); ctx.drawImage(img, x, y, imgWidth, imgHeight); imagesLoaded++; if (imagesLoaded === capturedDataUrls.length) { stripCanvas.style.display = 'block'; // Copia a tirinha diretamente para a área de transferência do iPad stripCanvas.toBlob(async (blob) => { try { const data = [new ClipboardItem({ [blob.type]: blob })]; await navigator.clipboard.write(data); statusText.innerText = "Tirinha copiada! Imprimindo..."; } catch (err) { console.error("Erro ao copiar imagem: ", err); } }); } }; img.src = dataUrl; }); }
